# Focused Windows Rust contract, run by scripts/windows-contract.py on a native Windows runner.
#
#   test <libtest-filter>            one `cargo test --lib <filter>` invocation (substring match)
#   skip <test-path> # <reason>      excluded from every invocation via libtest `--skip`
#
# Every skip is printed by the runner so unsupported coverage stays visible in CI logs. Remove a
# skip as soon as its test has a Windows-native implementation; never add one to hide a failure
# that a Windows regression would trigger.

# Self-update verification, checksum selection and crash-safe activation (src/cmd/update.rs).
test cli::update
# Platform process/window/path helpers, including the Windows-only branches of platform/mod.rs.
test platform::
# Background job process identity and Windows job-object group ownership.
test tools::background::procinfo
# Edit transactions exercise filesystem replacement on the native filesystem.
test tools::edit_transaction

# Process-group deadlines and pipe handling are exercised with POSIX `/bin/sh` fixtures. Windows
# needs its own job-object fixtures before these run here.
skip runtime::process::tests::captures_streams_with_caps_and_stdin # /bin/sh fixture; no Windows shell
skip runtime::process::tests::deadline_kills_the_whole_group # POSIX process-group fixture
skip runtime::process::tests::a_daemon_holding_the_pipe_does_not_hang_the_caller # /bin/sh daemon fixture
skip runtime::process::tests::cancellation_before_start_runs_nothing # /bin/sh fixture; no Windows shell
