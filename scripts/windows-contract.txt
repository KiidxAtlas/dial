# Focused Windows Rust contract, run by scripts/windows-contract.py on a native Windows runner.
#
#   test <libtest-filter>                  workspace library invocation (substring match)
#   test-bin <package> <binary> <filter>    explicit binary target invocation (substring match)
#   skip <test-path> # <reason>            excluded from every invocation via libtest `--skip`
#
# Every skip is printed by the runner so unsupported coverage stays visible in CI logs. Remove a
# skip as soon as its test has a Windows-native implementation; never add one to hide a failure
# that a Windows regression would trigger.

test platform::procinfo

# `tools::edit::edit_transaction` module has no tests of its own (unix, windows and mod alike).
test tools::edit::document_transaction
